	local function UpdateAccentColor(newColor)
		Colors.Accent = newColor
		UIStroke.Color = newColor
		SStroke.Color = newColor
		InfoStroke.Color = newColor
		Title.TextColor3 = newColor
		GlowFrame.ImageColor3 = newColor
		for _, data in pairs(AllUIElements) do
			if data.Type == "TabIndicator" or data.Type == "SliderFill" then
				data.Element.BackgroundColor3 = newColor
			elseif data.Type == "ValueLabel" then
				data.Element.TextColor3 = newColor
			elseif data.Type == "ScrollBar" then
				data.Element.ScrollBarImageColor3 = newColor
			end
		end
		local knobColor = GetContrastColor(newColor)
		for _, toggle in pairs(AllToggles) do
			if toggle.State then
				toggle.Button.BackgroundColor3 = newColor
				toggle.Circle.BackgroundColor3 = knobColor
			end
		end
		for _, tab in pairs(Tabs) do
			if tab.Frame.Visible then tab.Frame.BackgroundColor3 = newColor end
		end
	end

	local function IsValid(obj)
		if not obj or not obj.Parent then return false end
		local n = obj.Name:lower()
		if string.find(n, "whale") or string.find(n, "shark") or string.find(n, "mobywood")
			or string.find(n, "seal") or string.find(n, "piranha") or string.find(n, "kraken") then
			if obj:FindFirstChild("HumanoidRootPart") or obj.PrimaryPart then
				return true
			end
		end
		return false
	end

	local function UpdateESP()
		for _, obj in pairs(Workspace:GetChildren()) do
			if IsValid(obj) and not ESP_Storage[obj] then
				pcall(function()
					local hl = Instance.new("Highlight", TargetParent)
					hl.Adornee = obj
					hl.FillColor = Color3.fromRGB(255, 255, 255)
					hl.OutlineColor = Color3.fromRGB(0, 0, 0)
					hl.FillTransparency = 0.75
					local bb = Instance.new("BillboardGui", TargetParent)
					bb.Adornee = obj:FindFirstChild("HumanoidRootPart") or obj.PrimaryPart
					bb.Size = UDim2.new(0, 120, 0, 30)
					bb.StudsOffset = Vector3.new(0, 6, 0)
					bb.AlwaysOnTop = true
					local txt = Instance.new("TextLabel", bb)
					txt.Size = UDim2.new(1, 0, 1, 0)
					txt.BackgroundTransparency = 1
					txt.TextColor3 = Color3.new(1, 1, 1)
					txt.Font = Enum.Font.GothamBold
					txt.TextSize = 12
					txt.TextStrokeTransparency = 0
					ESP_Storage[obj] = {hl, bb, txt}
				end)
			end
		end
		for obj, data in pairs(ESP_Storage) do
			if obj.Parent then
				local root = obj:FindFirstChild("HumanoidRootPart") or obj.PrimaryPart
				if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") and root then
					local dist = (LocalPlayer.Character.HumanoidRootPart.Position - root.Position).Magnitude
					data[3].Text = obj.Name .. " [" .. math.floor(dist) .. "m]"
				end
			else
				pcall(function() data[1]:Destroy() data[2]:Destroy() end)
				ESP_Storage[obj] = nil
			end
		end
	end

	local function SendWebhook(fishName, rarity, coins)
		if not Settings.WebhookEnabled or Settings.WebhookURL == "" then return end
		local rarityOrder = {Common=1,Uncommon=2,Rare=3,Epic=4,Legendary=5,Mythic=6,Mutated=7}
		local minRank = rarityOrder[Settings.WebhookRarity] or 5
		if (rarityOrder[rarity] or 0) < minRank then return end
		task.spawn(function()
			pcall(function()
				local embed = {
					title   = "Rare Fish Caught!",
					color   = 3394815,
					fields  = {
						{name="Fish",   value=tostring(fishName), inline=true},
						{name="Rarity", value=tostring(rarity),   inline=true},
						{name="Coins",  value=tostring(coins),    inline=true},
					},
					footer    = {text="Ocean Hub v1 - Fishing Simulator"},
					timestamp = DateTime.now():ToIsoDate(),
				}
				local body = HttpService:JSONEncode({embeds={embed}})
				local req = (syn and syn.request) or (http and http.request) or http_request or request
				if req then
					req({
						Url     = Settings.WebhookURL,
						Method  = "POST",
						Headers = {["Content-Type"]="application/json"},
						Body    = body,
					})
				end
				SessionStats.RareCatches = SessionStats.RareCatches + 1
			end)
		end)
	end

	RunService.Heartbeat:Connect(function()
		if Settings.NoFog then
			Lighting.FogEnd = 8999999488
			Lighting.FogStart = 8999999488
			for _, v in pairs(Lighting:GetChildren()) do
				if v:IsA("Atmosphere") or v:IsA("Sky") then v:Destroy() end
			end
		end

		if Settings.ESP then
			UpdateESP()
		else
			for o, d in pairs(ESP_Storage) do
				pcall(function() d[1]:Destroy() d[2]:Destroy() end)
				ESP_Storage[o] = nil
			end
		end

		if Settings.WalkSpeed and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
			LocalPlayer.Character.Humanoid.WalkSpeed = Settings.WalkSpeedVal
		end

		CheckAndEquipTool()

		-- Player Fly
		if Settings.PlayerFly and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
			local root = LocalPlayer.Character.HumanoidRootPart
			if not PlayerFlyVel then
				PlayerFlyVel = Instance.new("BodyVelocity", root)
				PlayerFlyVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
			end
			local camCF = Camera.CFrame
			local moveDir = Vector3.zero
			if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + camCF.LookVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - camCF.LookVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + camCF.RightVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - camCF.RightVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.E) then moveDir = moveDir + Vector3.new(0, 1, 0) end
			if UserInputService:IsKeyDown(Enum.KeyCode.Q) then moveDir = moveDir - Vector3.new(0, 1, 0) end -- fixed: was referencing undefined "moveVec"
			PlayerFlyVel.Velocity = moveDir * Settings.FlySpeed
		elseif PlayerFlyVel then
			PlayerFlyVel:Destroy()
			PlayerFlyVel = nil
		end

		-- Mob Hunt / Boss Farm
		if (Settings.MobHunt or Settings.BossFarm) and LocalPlayer.Character then
			local root = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
			if root then
				if CurrentTarget and CurrentTarget.Parent then
					local targetRoot = CurrentTarget:FindFirstChild("HumanoidRootPart") or CurrentTarget.PrimaryPart
					if targetRoot then
						LastTargetPos = targetRoot.Position
						local currentHeight = (Settings.BossFarm and Settings.BossHuntHeight) or Settings.HuntHeight
						root.CFrame = CFrame.new(targetRoot.Position + Vector3.new(0, currentHeight, 0))
						root.AssemblyLinearVelocity = Vector3.zero
						SText.Text = "HUNTING: " .. CurrentTarget.Name
						StatusFrame.Visible = true
						LootingTime = 0
					end
				elseif LastTargetPos then
					if LootingTime == 0 then LootingTime = tick() + 1.5 end
					if tick() < LootingTime then
						local rayParams = RaycastParams.new()
						rayParams.FilterDescendantsInstances = {LocalPlayer.Character}
						rayParams.FilterType = Enum.RaycastFilterType.Exclude
						local rayResult = Workspace:Raycast(Vector3.new(LastTargetPos.X, LastTargetPos.Y + 30, LastTargetPos.Z), Vector3.new(0, -500, 0), rayParams)
						local landY = (rayResult and rayResult.Position.Y) or LastTargetPos.Y
						root.CFrame = root.CFrame:Lerp(CFrame.new(LastTargetPos.X, landY + 4, LastTargetPos.Z), 0.12)
						root.AssemblyLinearVelocity = Vector3.zero
						SText.Text = "COLLECTING..."
						StatusFrame.Visible = true
					else
						LastTargetPos = nil
						LootingTime = 0
					end
				else
					StatusFrame.Visible = false
					local minDist, found = math.huge, nil
					for _, obj in pairs(Workspace:GetChildren()) do
						if IsValid(obj) then
							local objRoot = obj:FindFirstChild("HumanoidRootPart") or obj.PrimaryPart
							if objRoot then
								local dist = (root.Position - objRoot.Position).Magnitude
								local isMoby = string.find(obj.Name:lower(), "mobywood")
								local isKraken = string.find(obj.Name:lower(), "kraken")
								if Settings.BossFarm then
									if (isMoby or isKraken) and dist < minDist then
										minDist = dist
										found = obj
									end
								elseif Settings.MobHunt then
									if not isMoby and not isKraken then
										if next(Settings.TargetList) == nil or Settings.TargetList[obj.Name] then
											if dist < minDist then
												minDist = dist
												found = obj
											end
										end
									end
								end
							end
						end
					end
					CurrentTarget = found
				end
			end
		else
			StatusFrame.Visible = false
		end

		-- Boat
		if LocalPlayer.Character then
			local hum = LocalPlayer.Character:FindFirstChild("Humanoid")
			if hum and hum.SeatPart and hum.SeatPart:IsA("VehicleSeat") then
				local seat = hum.SeatPart
				if Settings.BoatFly then
					InfoFrame.Visible = true
					Camera.CameraSubject = hum
					if not BoatFlyVel then
						BoatFlyVel = Instance.new("BodyVelocity", seat)
						BoatFlyVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
					end
					if not BoatFlyGyro then
						BoatFlyGyro = Instance.new("BodyGyro", seat)
						BoatFlyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
						BoatFlyGyro.D = 10
					end
					local camCF = Camera.CFrame
					local moveVec = Vector3.zero
					if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVec = moveVec + camCF.LookVector end
					if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVec = moveVec - camCF.LookVector end
					if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVec = moveVec - camCF.RightVector end
					if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVec = moveVec + camCF.RightVector end
					if UserInputService:IsKeyDown(Enum.KeyCode.E) then moveVec = moveVec + Vector3.new(0, 1, 0) end
					if UserInputService:IsKeyDown(Enum.KeyCode.Q) then moveVec = moveVec - Vector3.new(0, 1, 0) end
					BoatFlyVel.Velocity = moveVec * Settings.BoatSpeedVal
					BoatFlyGyro.CFrame = CFrame.new(seat.Position) * CFrame.Angles(0, math.atan2(camCF.LookVector.X, camCF.LookVector.Z), 0)
				else
					InfoFrame.Visible = false
					if BoatFlyVel then BoatFlyVel:Destroy() BoatFlyVel = nil end
					if BoatFlyGyro then BoatFlyGyro:Destroy() BoatFlyGyro = nil end
					if Settings.BoatSpeed then seat.MaxSpeed = Settings.BoatSpeedVal end
					if Settings.BoatAgility then
						seat.TurnSpeed = 5
						seat.Torque = math.huge
					end
				end
			else
				if BoatFlyVel then BoatFlyVel:Destroy() BoatFlyVel = nil end
				if BoatFlyGyro then BoatFlyGyro:Destroy() BoatFlyGyro = nil end
				InfoFrame.Visible = false
			end
		end
	end)
